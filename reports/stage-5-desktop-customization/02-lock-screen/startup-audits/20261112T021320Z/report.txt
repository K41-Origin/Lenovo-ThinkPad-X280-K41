    printf 'K41 lock startup: ERROR: %s\n' "$*" >&2

    if [[ -e $path ]]; then

lock_ipc_ready() {
    local listing
    if ! listing="$(timeout --signal=TERM --kill-after=1 "${command_timeout}s" caelestia shell -s 2>&1)"; then
        last_ipc_output=$listing
        return 1
    fi
    if grep -Eq 'target[[:space:]]+lock([[:space:]]|$)' <<<"$listing" &&
       grep -Eq 'function[[:space:]]+isLocked\(\):[[:space:]]*bool' <<<"$listing"; then
        return 0

    last_ipc_output=$listing
    return 1
}

    local state
    state="$(timeout --signal=TERM --kill-after=1 "${command_timeout}s" caelestia shell lock isLocked 2>&1)" ||

        log 'Caelestia lock IPC is ready.'

        log 'Caelestia lock did not reach verified state within the bounded verification window.'

        log 'Caelestia lock IPC rejected or timed out on the lock request.'

    log "Caelestia lock IPC did not become ready within ${ready_timeout}s."

        log "Last Caelestia IPC response: ${last_ipc_output:0:1200}"
    else
        log 'Caelestia IPC returned no diagnostic output.'
    fi

    die 'Caelestia lock was not verified and Hyprlock fallback is unavailable.'

    die "Managed Hyprlock fallback config is missing or unsafe: $hyprlock_config"
[[ $(stat -c %u -- "$hyprlock_config") == "$EUID" ]] ||
    die 'Managed Hyprlock fallback config is not owned by the desktop user.'

if ! jobs -pr | grep -Fxq "$lock_pid"; then
    set +e
    wait "$lock_pid"
    fallback_status=$?
    set -e
    die "Hyprlock exited before readiness could be established (exit=$fallback_status)."
fi

log "Hyprlock fallback is running (pid=$lock_pid); waiting for unlock/exit."
set +e
wait "$lock_pid"
fallback_status=$?
set -e
log "Hyprlock fallback exited (exit=$fallback_status)."
exit "$fallback_status"

#!/usr/bin/env bash
# Wait for Caelestia's lock IPC, request a lock, verify it, then dismiss K41 overlay.
set -Eeuo pipefail

die() { printf 'K41 Caelestia lock: %s\n' "$*" >&2; exit 1; }
mark_ready() {
    local marker="$HOME/.cache/k41/boot-lock-ready"
    install -d -m 0700 "$(dirname -- "$marker")"
    : >"$marker"
    chmod 0600 "$marker"
}
dismiss_splash() {
    local pid_file="$HOME/.cache/k41/boot-splash.pid" pid
    if [[ -f $pid_file && ! -L $pid_file ]]; then
        IFS= read -r pid <"$pid_file" || true
        if [[ $pid =~ ^[1-9][0-9]*$ ]] && kill -0 "$pid" 2>/dev/null; then
            kill "$pid" 2>/dev/null || true
        fi
        rm -f -- "$pid_file"
    fi
}
command -v caelestia >/dev/null 2>&1 || die 'caelestia CLI is unavailable.'
command -v sleep >/dev/null 2>&1 || die 'sleep is unavailable.'
command -v grep >/dev/null 2>&1 || die 'grep is unavailable.'

attempts=${K41_LOCK_RETRIES:-45}
interval=${K41_LOCK_INTERVAL:-1}
verify_attempts=${K41_LOCK_VERIFY_RETRIES:-10}
[[ $attempts =~ ^[1-9][0-9]*$ ]] || die 'K41_LOCK_RETRIES must be a positive integer.'
[[ $interval =~ ^[0-9]+([.][0-9]+)?$ ]] || die 'K41_LOCK_INTERVAL must be numeric.'
[[ $verify_attempts =~ ^[1-9][0-9]*$ ]] || die 'K41_LOCK_VERIFY_RETRIES must be a positive integer.'

lock_ipc_ready() {
    caelestia shell -s 2>/dev/null |
        grep -Eq 'target[[:space:]]+lock([[:space:]]|$)' &&
    caelestia shell -s 2>/dev/null |
        grep -Eq 'function[[:space:]]+isLocked\(\):[[:space:]]*bool'
}
is_locked() {
    local state
    state=$(caelestia shell lock isLocked 2>/dev/null) || return 1
    [[ $state =~ ^[[:space:]]*true[[:space:]]*$ ]]
}
for ((i = 1; i <= attempts; i++)); do
    if lock_ipc_ready && caelestia shell lock lock; then
        for ((v = 1; v <= verify_attempts; v++)); do
            if is_locked; then
                printf 'K41 Caelestia lock: lock state verified.\n'
                dismiss_splash
                mark_ready
                exit 0
            fi
            sleep "$interval"
        done
    fi
    sleep "$interval"
done

printf 'K41 Caelestia lock: ERROR: lock IPC did not reach a verified locked state; starting Hyprlock fallback.\n' >&2
if command -v hyprlock >/dev/null 2>&1; then
    hyprlock &
    lock_pid=$!
    sleep 1
    if kill -0 "$lock_pid" 2>/dev/null; then
        dismiss_splash
        mark_ready
        wait "$lock_pid"
        exit $?
    fi
    wait "$lock_pid" || true
fi
die 'Caelestia lock could not be verified and Hyprlock fallback is unavailable.'
